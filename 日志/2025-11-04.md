public class FusionRunnerController : MonoBehaviour, INetworkRunnerCallbacks
{
    [Header("Refs")]
    public NetworkRunner runner;

    [Header("Arena Settings")]
    [Tooltip("Build Settings 里 Arena 场景的 BuildIndex")]
    public string arenaSceneName = "Arena";
    // Host/Client 启动事件：true=已启动, false=停止
    public event Action<bool> OnRunnerRunningChanged;
    static int _seq = 0;

    bool _isLoadingArena = false;
   
    // 内部状态
   
    int _pendingDurationSeconds = 0;

    void Awake()
    {
        if (!runner) runner = GetComponent<NetworkRunner>();
        DontDestroyOnLoad(gameObject);
        runner.AddCallbacks(this);
    }

    // ========== Host / Client 启动 ==========
    public async void StartAsHost(string room, Action<string> log = null)
    {
        log?.Invoke("启动 Host...");
        int arenaIndex = SceneUtility.GetBuildIndexByScenePath(arenaSceneName);
        // Host
        var result = await runner.StartGame(new StartGameArgs
        {
            GameMode = GameMode.Host,
            SessionName = string.IsNullOrWhiteSpace(room) ? "001" : room,
            SceneManager = gameObject.AddComponent<NetworkSceneManagerDefault>(),
            // ✅ 显式声明会话的起始场景（用当前激活场景的 buildIndex）
            //Scene = SceneRef.None
            Scene = SceneRef.FromIndex(arenaIndex)
            //Scene = SceneRef.FromIndex(SceneManager.GetActiveScene().buildIndex)
        });

        log?.Invoke(result.Ok ? "Host 启动成功" : $"Host 失败：{result.ShutdownReason}");
        OnRunnerRunningChanged?.Invoke(result.Ok);
    }
